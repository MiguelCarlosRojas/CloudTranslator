Visión General y Reglas de Conexión

Condiciones y formatos requeridos para interactuar con la API REST.

[ 1.1 // CANONICAL_ENDPOINT ]

URL Canónica del Endpoint

Cualquier aplicación web o móvil debe apuntar sus consultas HTTP hacia la ruta canónica en Render:

Endpoint de Traducción 
https://cloudtranslator.onrender.com/api/translate
[ 1.2 // NETWORK_RULES_CORS ]

Regla de Red Obligatoria y CORS Universal

El método siempre debe ser POST con cabecera Content-Type: application/json. Soporta CORS universal (Access-Control-Allow-Origin: *), por lo que puedes llamarlo directamente desde el navegador de tus usuarios sin proxies intermediarios.

Optimización Crítica: No envíes peticiones individuales por cada párrafo. Agrupa todos los textos en el arreglo texts: [...] para que el navegador solo haga 1 sola llamada de red.

[ 2.2 // JAVASCRIPT_SCRIPT ]

Script JavaScript Completo de Integración

Copia y pega este script en tu sistema. Consolida los textos en 1 sola petición HTTP, guarda la cookie cloud_translator_lang y recuerda el idioma al recargar la página:

Script JavaScript (Listo para Producción) 
// Configuración del Endpoint oficial de CloudTranslator
const API_TRANSLATE_URL = "https://cloudtranslator.onrender.com/api/translate";

// Mapas en memoria para restaurar y evitar consultas repetidas
const originalDomTexts = new Map();
const localCache = {}; // { 'en': { 'Hola': 'Hello' } }

// Helper para cookies de 365 días
function setLanguageCookie(lang) {
  document.cookie = `cloud_translator_lang=${lang}; path=/; max-age=31536000; SameSite=Lax`;
}

function getLanguageCookie() {
  const match = document.cookie.match(new RegExp('(^| )cloud_translator_lang=([^;]+)'));
  return match ? match[2] : null;
}

// 1. Escanear textos visibles ignorando scripts, estilos e inputs
function scanPageTexts() {
  if (originalDomTexts.size > 0) return;
  const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT, {
    acceptNode: (node) => {
      const parent = node.parentElement;
      if (!parent) return NodeFilter.FILTER_REJECT;
      const tag = parent.tagName.toLowerCase();
      if (['script', 'style', 'noscript', 'textarea', 'code', 'pre'].includes(tag)) {
        return NodeFilter.FILTER_REJECT;
      }
      return node.nodeValue.trim().length > 0 ? NodeFilter.FILTER_ACCEPT : NodeFilter.FILTER_REJECT;
    }
  });

  let currentNode;
  while ((currentNode = walker.nextNode())) {
    originalDomTexts.set(currentNode, currentNode.nodeValue);
  }
}

// 2. Traduce toda la página en 1 SOLA petición HTTP en segundo plano
async function translateSystemPage(targetLang) {
  const statusEl = document.getElementById("status-indicator");
  if (statusEl) statusEl.textContent = `Traduciendo (${targetLang.toUpperCase()})...`;

  scanPageTexts();
  setLanguageCookie(targetLang);

  if (!localCache[targetLang]) localCache[targetLang] = {};

  // Extraer cadenas únicas no cacheadas
  const uniqueStrings = [];
  originalDomTexts.forEach((originalText) => {
    const trimmed = originalText.trim();
    if (trimmed && !localCache[targetLang][trimmed] && !uniqueStrings.includes(trimmed)) {
      uniqueStrings.push(trimmed);
    }
  });

  // Si todas ya están en caché local, aplicar de inmediato (0ms de red)
  if (uniqueStrings.length === 0) {
    applyTranslationsFromCache(targetLang);
    if (statusEl) statusEl.textContent = `Listo (${targetLang.toUpperCase()})`;
    return;
  }

  try {
    // 1 SOLA PETICIÓN HTTP con todos los textos consolidados
    const response = await fetch(API_TRANSLATE_URL, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        texts: uniqueStrings,
        targetLanguage: targetLang
      })
    });

    if (!response.ok) throw new Error("Error en el microservicio: HTTP " + response.status);

    const data = await response.json();
    const translatedList = data.translatedTexts || [];

    // Guardar en la caché local
    uniqueStrings.forEach((originalStr, index) => {
      localCache[targetLang][originalStr] = translatedList[index] || originalStr;
    });

    // Reemplazar textos en el DOM
    applyTranslationsFromCache(targetLang);
    if (statusEl) statusEl.textContent = `Listo (${targetLang.toUpperCase()})`;
  } catch (error) {
    console.error("Fallo al conectar con CloudTranslator:", error);
    if (statusEl) statusEl.textContent = "Error de conexión";
  }
}

// 3. Aplicar textos reemplazando nodos sin romper la estructura HTML
function applyTranslationsFromCache(targetLang) {
  originalDomTexts.forEach((originalText, textNode) => {
    const trimmed = originalText.trim();
    const translated = localCache[targetLang] ? localCache[targetLang][trimmed] : null;
    if (translated) {
      textNode.nodeValue = originalText.replace(trimmed, translated);
    }
  });
}

// 4. Restaurar al idioma original de la página
function restoreOriginalLanguage() {
  originalDomTexts.forEach((originalText, textNode) => {
    textNode.nodeValue = originalText;
  });
  setLanguageCookie("original");
  const statusEl = document.getElementById("status-indicator");
  if (statusEl) statusEl.textContent = "Original restaurado";
}

// 5. Asignación de Eventos y Carga Automática por Cookie
document.addEventListener("DOMContentLoaded", () => {
  // Botón directo a inglés
  const btnEn = document.getElementById("btn-english");
  if (btnEn) btnEn.addEventListener("click", () => translateSystemPage("en"));

  // Botón con código dinámico
  const btnCustom = document.getElementById("btn-translate-lang");
  if (btnCustom) {
    btnCustom.addEventListener("click", () => {
      const code = (document.getElementById("lang-code").value || "").trim().toLowerCase();
      if (code) translateSystemPage(code);
    });
  }

  // Botón restaurar
  const btnRestore = document.getElementById("btn-restore");
  if (btnRestore) btnRestore.addEventListener("click", restoreOriginalLanguage);

  // Auto-traducción en 0ms leyendo la Cookie al entrar o recargar
  const savedLang = getLanguageCookie();
  if (savedLang && savedLang !== "original") {
    translateSystemPage(savedLang);
  }
});

[ 3.1 // BACKEND_NODEJS ]

Conectar con Node.js / Express (Fetch o Axios)

Envía un arreglo de textos a traducir directamente desde tu servidor Node.js:

JavaScript (Node.js 18+) 
async function traducirTextos(textosArray, idiomaDestino = "en") {
  const url = "https://cloudtranslator.onrender.com/api/translate";
  
  const respuesta = await fetch(url, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      texts: textosArray,
      targetLanguage: idiomaDestino
    })
  });

  if (!respuesta.ok) {
    throw new Error(`Fallo HTTP: ${respuesta.status}`);
  }

  const datos = await respuesta.json();
  return datos.translatedTexts; // Retorna array de strings traducidos
}

// Ejemplo de uso:
traducirTextos(["Iniciar sesión", "Cerrar sesión"], "en")
  .then(resultado => console.log(resultado)) // ["Log in", "Sign out"]
  .catch(err => console.error(err));
[ 3.2 // BACKEND_PYTHON ]

Conectar con Python (Requests / HTTPX)

Integra CloudTranslator en scripts de Python, Django o FastAPI:

Python 3 (requests) 
import requests

def traducir_textos(textos, target_lang="en"):
    url = "https://cloudtranslator.onrender.com/api/translate"
    payload = {
        "texts": textos,
        "targetLanguage": target_lang
    }
    
    response = requests.post(url, json=payload, headers={"Content-Type": "application/json"})
    response.raise_for_status()
    
    data = response.json()
    return data.get("translatedTexts", [])

# Ejemplo de uso:
resultado = traducir_textos(["Panel de control", "Configuración de cuenta"], target_lang="en")
print(resultado)  # ['Dashboard', 'Account Settings']
[ 3.3 // BACKEND_PHP ]

Conectar con PHP (cURL nativo)

Consúmelo en aplicaciones PHP tradicionales, Laravel o WordPress:

PHP (cURL) 
<?php
function traducirTextos(array $textos, string $idioma = "en"): array {
    $url = "https://cloudtranslator.onrender.com/api/translate";
    $payload = json_encode([
        "texts" => $textos,
        "targetLanguage" => $idioma
    ]);

    $ch = curl_init($url);
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    curl_setopt($ch, CURLOPT_POST, true);
    curl_setopt($ch, CURLOPT_POSTFIELDS, $payload);
    curl_setopt($ch, CURLOPT_HTTPHEADER, [
        "Content-Type: application/json"
    ]);

    $response = curl_exec($ch);
    $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    curl_close($ch);

    if ($httpCode !== 200) {
        throw new Exception("Error en CloudTranslator con código HTTP: " . $httpCode);
    }

    $data = json_decode($response, true);
    return $data["translatedTexts"] ?? [];
}

// Ejemplo de uso:
$traducciones = traducirTextos(["Bienvenido a nuestra tienda", "Ver catálogo"], "en");
print_r($traducciones);
?>

[ S.04 // PERFORMANCE_COOKIES ]

Persistencia por Cookies y Rendimiento de Red

Cómo funciona la cookie técnica y el comportamiento de la infraestructura en Render.

[ 4.1 // COOKIE_SPECIFICATION ]

Cookie Técnica cloud_translator_lang

Al seleccionar un idioma o presionar el botón de inglés, el cliente establece la cookie técnica:

Encabezado Set-Cookie
Set-Cookie: cloud_translator_lang=en; Path=/; Max-Age=31536000; SameSite=Lax
  • Guarda estrictamente el código del idioma deseado (ej: "en", "fr", "de").
  • Tiene vigencia de 365 días para no requerir configuración repetida en cada visita.
  • Permite que el navegador restaure el idioma inmediatamente en 0ms tras recargar la página.
[ 4.2 // RENDER_COLD_START ]

Comportamiento de la Capa Gratuita de Render

El microservicio se ejecuta en el clúster gratuito de Render:

Arranque en Frío: Si no hay tráfico entrante durante 15 minutos, el contenedor entra en modo de reposo. La primera consulta reactiva el servidor en aproximadamente 25 a 35 segundos. Las peticiones subsecuentes responden con latencia promedio de 150 a 350 milisegundos.

[ 5.2 // AUDIO_SYNTHESIS ]

Generación y Reproducción Directa en Frontend

Envía el texto a POST /api/tts. El backend responde con el audio en base64 y URLs de streaming para reproducción inmediata:

JavaScript - Síntesis y Reproducción HTML5 
async function hablar(texto, voiceId = "es-ES-sofia") {
  const resp = await fetch("https://cloudtranslator.onrender.com/api/tts", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      text: texto,
      voiceId: voiceId,
      lang: "es-ES",
      rate: 1.0,
      pitch: 1.0
    })
  });

  const data = await resp.json();
  // Reproducir directamente el audio base64 devuelto por el servidor
  const audio = new Audio(data.audioBase64);
  audio.play();
}

Ejemplo en Python (Descarga a Archivo WAV)

Python 3 - Síntesis y Guardado en Disco 
import requests, base64

url = "https://cloudtranslator.onrender.com/api/tts"
payload = {
    "text": "Hola, este audio fue sintetizado mediante la API de CloudTranslator",
    "voiceId": "es-ES-sofia",
    "lang": "es-ES"
}

resp = requests.post(url, json=payload)
data = resp.json()

# Decodificar y guardar el archivo de audio localmente
audio_data = base64.b64decode(data["audioBase64"].split(",")[1])

    f.write(audio_data)
print("Archivo guardado: locucion.wav")
[ S.06 // VOICE_CLONE_INTEGRATION ]

Pilar 3: Conexión al Motor de Clonación de Voz (IA)

Síntesis neuronal zero-shot recreando el timbre, formantes y entonación a partir de una muestra de voz.

[ 6.1 // SAMPLE_RECORDING ]

Muestra de Audio de Referencia

El sistema requiere una muestra de 4 a 10 segundos de la voz del usuario (formato .wav, .mp3, .ogg o grabada por micrófono). La muestra debe enviarse en formato Base64 con prefijo data:audio/....

Recomendación Acústica: Para una clonación óptima, habla con voz clara a unos 15 cm del micrófono sin música de fondo ni eco ambiental.
[ 6.2 // ZERO_SHOT_SYNTHESIS ]

Síntesis con Voz Clonada

Envía el guion deseado junto con la muestra base hacia POST /api/voice-clone:

JavaScript - Clonación y Reproducción Inmediata 
async function clonarVoz(muestraBase64, guionTexto) {
  const resp = await fetch("https://cloudtranslator.onrender.com/api/voice-clone", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      text: guionTexto,
      audioSampleBase64: muestraBase64,
      similarity: "high",
      pitch: 1.0
    })
  });

  const data = await resp.json();
  // Reproducir el audio sintetizado con el timbre del usuario
  const player = new Audio(data.audioBase64);
  player.play();

  // Enlace físico de descarga directa
  console.log("Enlace de descarga:", data.downloadUrl);
}

Ejemplo en Node.js

Node.js - Clonación y Guardado de Archivo 
const fs = require("fs");

async function clonarVozEnServidor() {
  const audioSampleBuffer = fs.readFileSync("./mi-voz-muestra.wav");
  const audioSampleBase64 = `data:audio/wav;base64,${audioSampleBuffer.toString("base64")}`;

  const response = await fetch("https://cloudtranslator.onrender.com/api/voice-clone", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      text: "Mensaje generado automáticamente con mi timbre de voz.",
      audioSampleBase64: audioSampleBase64,
      similarity: "high"
    })
  });

  const data = await response.json();
  const rawWav = Buffer.from(data.audioBase64.split(",")[1], "base64");
  fs.writeFileSync("./voz-clonada-resultado.wav", rawWav);
  console.log("Archivo sintetizado con éxito: voz-clonada-resultado.wav");
}

Pilar 4: Conversor Multimedia Universal (MP3 y MP4)

Motor universal de extracción y transcodificación de audio y video en la nube. Permite convertir enlaces de YouTube, TikTok, Instagram Reels, Facebook, Twitter/X, Pinterest y Reddit a archivos de audio MP3 (hasta 320 kbps) o video MP4 sin marcas de agua, con soporte para streaming HTTP 206 y descarga física directa.

7.1 Detección de Plataformas & Extracción de Metadatos

El endpoint POST /api/media/info analiza cualquier URL, detecta el servicio de procedencia, valida el identificador y extrae título, autor, duración, miniatura en alta resolución y las opciones de bitrate disponibles.

Petición cURL para Inspeccionar Enlace

cURL - POST /api/media/info 
curl -X POST https://cloudtranslator.onrender.com/api/media/info \
  -H "Content-Type: application/json" \
  -d '{
    "url": "https://www.youtube.com/watch?v=jNQXAC9IVRw"
  }'

7.2 Conversión a MP3 / MP4, Streaming y Descarga

El endpoint POST /api/media/convert procesa el enlace con el formato solicitado ("mp3" o "mp4") y la calidad deseada ("320k", "192k", "1080p", etc.). Retorna tokens seguros para streaming en tiempo real y descarga física.

Petición en JavaScript (Fetch)

JavaScript - Conversión y Reproducción 
async function convertirVideo(enlaceVideo, formato = "mp3", calidad = "320k") {
  const res = await fetch("https://cloudtranslator.onrender.com/api/media/convert", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      url: enlaceVideo,
      format: formato,
      quality: calidad
    })
  });

  const data = await res.json();
  console.log("Título:", data.title);
  console.log("Tamaño:", data.sizeFormatted);

  // Reproducir en navegador sin esperar la descarga completa
  if (data.format === "mp3") {
    const audio = new Audio(data.streamUrl);
    audio.play();
  }

  // Enlace físico directo para descargar el archivo
  window.location.href = data.downloadUrl;
}
